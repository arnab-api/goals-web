def count_and_say(terms):↵
    if terms == 0:↵
        return ""↵
    result = ["1"]↵
    for _ in range(1, terms):↵
        current_term = []↵
        count = 1↵
        for i in range(1, len(result)):↵
            if result[i] == result[i - 1]:↵
                count += 1↵
            else:↵
                current_term.append(str(count))↵
                current_term.append(result[i - 1])↵
                count = 1↵
        current_term.append(str(count))↵
        current_term.append(result[-1])↵
        result = current_term↵
    return ''.join(result)↵
↵
# Example usage:↵
print(count_and_say(5))  # Output: "111221"↵
