def min_spanning_tree_kruskal(edges, num_vertices):↵
    def find(parent, i):↵
        if parent[i] == i:↵
            return i↵
        return find(parent, parent[i])↵
↵
    def union(parent, rank, x, y):↵
        rootX = find(parent, x)↵
        rootY = find(parent, y)↵
↵
        if rank[rootX] > rank[rootY]:↵
            parent[rootY] = rootX↵
        elif rank[rootX] < rank[rootY]:↵
            parent[rootX] = rootY↵
        else:↵
            parent[rootY] = rootX↵
            rank[rootX] += 1↵

    rank = [0] * num_vertices↵
    mst = []↵

    for u, v, weight in edges:↵
        if find(parent, u) != find(parent, v):↵
            union(parent, rank, u, v)↵
            mst.append((u, v, weight))↵
↵
    return mst↵
