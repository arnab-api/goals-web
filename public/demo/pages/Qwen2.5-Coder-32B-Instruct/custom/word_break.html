def word_break(phrase, dictionary):↵
    dp = [False] * (len(phrase) + 1)↵
    dp[0] = True↵
↵
    for i in range(1, len(phrase) + 1):↵
        for j in range(i):↵
            if dp[j] and phrase[j:i] in dictionary:↵
                dp[i] = True↵
                break↵
↵
    return dp[len(phrase)]↵
↵
# Example usage:↵
dictionary = {"i", "like", "sam", "sung", "samsung", "mobile", "ice", "cream", "icecream", "man", "go", "mango"}↵
