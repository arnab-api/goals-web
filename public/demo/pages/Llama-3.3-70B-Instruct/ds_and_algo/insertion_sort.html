def insertion_sort(arry):↵
    for i in range(1, len(arry)):↵
        key = arry[i]↵
        j = i - 1↵
        while j >= 0 and arry[j] > key:↵
            arry[j + 1] = arry[j]↵
            j -= 1↵
        arry[j + 1] = key↵
    return arry↵
↵
