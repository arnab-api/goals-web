    def heapify(ary, n, i):↵
        largest = i↵
        left = 2 * i + 1↵
        right = 2 * i + 2↵
        if left < n and ary[i] < ary[left]:↵
            largest = left↵
        if right < n and ary[largest] < ary[right]:↵

        if largest!= i:↵

    for i in range(n // 2 - 1, -1, -1):↵
