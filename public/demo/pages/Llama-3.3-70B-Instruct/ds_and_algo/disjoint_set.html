Implement a disjoint set (union-find) with path compression and union by rank in Python<|eot_id|><|start_header_id|>assistant<|end_header_id|>↵
↵
<code>↵
class DisjointSet: ↵
    def __init__(self, n):↵
        self.parent = list(range(n))↵
        self.rank = [0] * n↵
↵
    def find(self, x):↵
        if self.parent[x]!= x:↵
            self.parent[x] = self.find(self.parent[x])↵
        return self.parent[x]↵
↵
    def union(self, x, y):↵
        root_x = self.find(x)↵
        root_y = self.find(y)↵
        if root_x!= root_y:↵
            if self.rank[root_x] > self.rank[root_y]:↵
                self.parent[root_y] = root_x↵
            elif self.rank[root_x] < self.rank[root_y]:↵
                self.parent[root_x] = root_y↵
            else:↵
                self.parent[root_y] = root_x↵
                self.rank[root_x] += 1↵
↵
    def connected(self, x, y):↵
        return self.find(x) == self.find(y)↵
↵
# Example usage:↵
ds = DisjointSet(10)↵
ds.union(1, 3)↵
ds.union(3, 5)↵
print(ds.connected(1, 5))  # True↵
print(ds.connected(1, 7))  # False↵
