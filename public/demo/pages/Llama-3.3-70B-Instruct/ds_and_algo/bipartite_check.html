def bipartite_check(Adj):↵
    color = {}↵
    for node in Adj:↵
        if node not in color:↵
            stack = [node]↵
            color[node] = 0↵
            while stack:↵
                node = stack.pop()↵
                for neighbor in Adj[node]:↵
                    if neighbor not in color:↵
                        stack.append(neighbor)↵
                        color[neighbor] = 1 - color[node]↵
                    elif color[neighbor] == color[node]:↵
                        return False↵
    return True↵
