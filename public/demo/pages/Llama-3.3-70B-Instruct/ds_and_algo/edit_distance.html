def edit_distance(phrase1, phrase2):↵
    m, n = len(phrase1), len(phrase2)↵
    dp = [[0] * (n + 1) for _ in range(m + 1)]↵
↵
    for i in range(m + 1):↵
        dp[i][0] = i↵
    for j in range(n + 1):↵
        dp[0][j] = j↵
↵
    for i in range(1, m + 1):↵
        for j in range(1, n + 1):↵
            cost = 0 if phrase1[i - 1] == phrase2[j - 1] else 1↵
            dp[i][j] = min(dp[i - 1][j] + 1, dp[i][j - 1] + 1, dp[i - 1][j - 1] + cost)↵
↵
    return dp[m][n]↵
↵
# Example usage:↵
phrase1 = "kitten"↵
phrase2 = "sitting"↵
print(edit_distance(phrase1, phrase2))↵
