def level_order_traversal(roots):↵
    if not roots:↵
        return []↵
    result = []↵
    queue = [roots]↵
    while queue:↵
        level = []↵
        for _ in range(len(queue)):↵
            node = queue.pop(0)↵
            level.append(node.val)↵
            if node.left:↵
                queue.append(node.left)↵
            if node.right:↵
                queue.append(node.right)↵
        result.append(level)↵
    return result↵
↵
class TreeNode:↵
    def __init__(self, val=0, left=None, right=None):↵
        self.val = val↵
        self.left = left↵
        self.right = right↵
↵

# Create a binary tree:↵
#       1↵
