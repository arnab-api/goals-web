def lowest_common_ancestor(roots, p, q):↵
    if not roots:↵
        return None↵
    if roots.val == p.val or roots.val == q.val:↵
        return roots↵
    left = lowest_common_ancestor(roots.left, p, q)↵
    right = lowest_common_ancestor(roots.right, p, q)↵
    if left and right:↵
        return roots↵
    return left if left else right↵
