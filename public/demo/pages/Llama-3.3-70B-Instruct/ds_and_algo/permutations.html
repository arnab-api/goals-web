def permutations(lst):↵
    if len(lst) == 1:↵
        return [lst]↵
    result = []↵
    for i in range(len(lst)):↵
        m = lst[i]↵
        remaining = lst[:i] + lst[i+1:]↵
        for p in permutations(remaining):↵
            result.append([m] + p)↵
    return result↵
