def longest_increasing_subsequence_bs(sequences):↵
    if not sequences:↵
        return 0↵

    dp = [sequences[0]]↵

def binary_search(arr, target):↵
    left, right = 0, len(arr) - 1↵
    while left < right:↵
        mid = (left + right) // 2↵
        if arr[mid] < target:↵

        else:↵
            right = mid↵
    return left↵
