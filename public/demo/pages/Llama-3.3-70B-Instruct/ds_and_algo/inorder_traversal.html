def inorder_traversal(root):↵
    stack = []↵
    result = []↵
    while True:↵
        if root is not None:↵
            stack.append(root)↵
            root = root.left↵
        elif stack:↵
            node = stack.pop()↵
            result.append(node.val)↵
            root = node.right↵
        else:↵
            break↵
    return result↵
↵

        self.right = right↵
