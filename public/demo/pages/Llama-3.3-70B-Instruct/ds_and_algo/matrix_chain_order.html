def matrix_chain_order(orders):↵
    n = len(orders) - 1↵
    m = [[0 for _ in range(n)] for _ in range(n)]↵
    s = [[0 for _ in range(n)] for _ in range(n)]↵
↵
    for i in range(n):↵
        m[i][i] = 0↵
↵
    for L in range(2, n + 1):↵
        for i in range(n - L + 1):↵
            j = i + L - 1↵
            m[i][j] = float('inf')↵
            for k in range(i, j):↵
                q = m[i][k] + m[k + 1][j] + orders[i] * orders[k + 1] * orders[j + 1]↵
                if q < m[i][j]:↵
                    m[i][j] = q↵
                    s[i][j] = k↵
↵
    return m, s↵

def optimal_parens(s, i, j):↵

    else:↵
        return f"({optimal_parens(s, i, s[i][j])}{optimal_parens(s, s[i][j] + 1, j)})"↵

def matrix_chain_multiplication(orders):↵
    m, s = matrix_chain_order(orders)↵
    return m[0][-1], optimal_parens(s, 0, len(orders) - 2)↵
