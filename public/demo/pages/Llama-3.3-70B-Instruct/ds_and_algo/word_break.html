def word_break(phrase, word_dict):↵
    dp = [False] * (len(phrase) + 1)↵
    dp[0] = True↵
    for i in range(1, len(phrase) + 1):↵
        for j in range(i):↵
            if dp[j] and phrase[j:i] in word_dict:↵
                dp[i] = True↵
                break↵
    return dp[-1]↵

def word_break_ii(phrase, word_dict):↵
    memo = dict()↵
    def dfs(phrase):↵
        if phrase in memo: return memo[phrase]↵
        if not phrase: return []↵
        res = []↵
        for word in word_dict:↵
            if not phrase.startswith(word):↵
                continue↵
            if len(word) == len(phrase):↵
                res.append(word)↵
            else:↵
                result_of_rest = dfs(phrase[len(word):])↵
                for item in result_of_rest:↵
                    item = word +'' + item↵
                    res.append(item)↵
        memo[phrase] = res↵
        return res↵
    return dfs(phrase)↵
