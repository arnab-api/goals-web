def path_sum(roots, target):↵
    if not roots:↵
        return False↵
    def dfs(node, target):↵
        if not node:↵
            return False↵
        if not node.left and not node.right and node.val == target:↵
            return True↵
        return dfs(node.left, target - node.val) or dfs(node.right, target - node.val)↵
    return dfs(roots, target)↵
↵
class TreeNode:↵
    def __init__(self, val=0, left=None, right=None):↵
        self.val = val↵
        self.left = left↵
        self.right = right↵
↵
# Example usage:↵
# Create a binary tree↵
#       5↵
#      / \↵
#     4   8↵
#    /   / \↵
#   11  13  4↵
#  / \       \↵
# 7   2       5↵
↵
