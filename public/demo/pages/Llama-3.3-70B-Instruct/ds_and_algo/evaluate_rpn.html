def evaluate_rpn(operations):↵
    stack = []↵
    for op in operations:↵
        if op in {'+', '-', '*', '/'}:↵
            b = stack.pop()↵
            a = stack.pop()↵
            if op == '+':↵
                stack.append(a + b)↵
            elif op == '-':↵
                stack.append(a - b)↵
            elif op == '*':↵
                stack.append(a * b)↵
            elif op == '/':↵
                stack.append(int(a / b))↵
        else:↵
            stack.append(int(op))↵
    return stack[0]↵
↵
# Example usage:↵
print(evaluate_rpn(["2","1","+","3","*"]))  # Output: 9↵
