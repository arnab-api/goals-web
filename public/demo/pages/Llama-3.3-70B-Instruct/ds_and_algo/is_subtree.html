def is_subtree(roots, subtrees):↵
    def is_match(t1, t2):↵
        if t1 is None and t2 is None:↵
            return True↵
        if t1 is None or t2 is None:↵
            return False↵
        return (t1.val == t2.val and ↵
                is_match(t1.left, t2.left) and ↵
                is_match(t1.right, t2.right))↵

    return traverse(roots, subtrees)↵
