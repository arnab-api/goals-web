def selection_sort(arry):↵
    for i in range(len(arry)):↵
        min_index = i↵
        for j in range(i+1, len(arry)):↵
            if arry[j] < arry[min_index]:↵
                min_index = j↵
        arry[i], arry[min_index] = arry[min_index], arry[i]↵
    return arry↵
