def bellman_ford(grafo, origen):↵
    distancia = {vertice: float('inf') for vertice in grafo}↵
    distancia[origen] = 0↵
    anterior = {vertice: None for vertice in grafo}↵
↵
    for _ in range(len(grafo) - 1):↵
        for vertice in grafo:↵
            for vecino, peso in grafo[vertice].items():↵
                if distancia[vertice] + peso < distancia[vecino]:↵
                    distancia[vecino] = distancia[vertice] + peso↵
                    anterior[vecino] = vertice↵
↵
    for vertice in grafo:↵
        for vecino, peso in grafo[vertice].items():↵
            if distancia[vertice] + peso < distancia[vecino]:↵
                raise ValueError("Grafo contiene un ciclo negativo")↵
↵
    return distancia, anterior↵
↵
